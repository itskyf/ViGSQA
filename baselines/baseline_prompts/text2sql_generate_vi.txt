--- Query Templates (use exactly these patterns) ---

KNN – find nearest POI by name:
  SELECT p.poi_name
  FROM pois p
  CROSS JOIN (SELECT geometry FROM pois WHERE poi_name ILIKE '%<ref>%' LIMIT 1) ref
  WHERE p.<category_col> ILIKE '%<value>%'
  ORDER BY p.geometry <-> ref.geometry LIMIT 1

KNN – find nearest POI location (lon/lat):
  SELECT p.poi_name,
         ST_X(ST_Centroid(p.geometry::geometry)) AS lon,
         ST_Y(ST_Centroid(p.geometry::geometry)) AS lat
  FROM pois p
  CROSS JOIN (SELECT geometry FROM pois WHERE poi_name ILIKE '%<ref>%' LIMIT 1) ref
  WHERE p.<category_col> ILIKE '%<value>%'
  ORDER BY p.geometry <-> ref.geometry LIMIT 1

KNN – distance to nearest POI:
  SELECT p.poi_name, ST_Distance(p.geometry, ref.geometry) AS dist_m
  FROM pois p
  CROSS JOIN (SELECT geometry FROM pois WHERE poi_name ILIKE '%<ref>%' LIMIT 1) ref
  WHERE p.<category_col> ILIKE '%<value>%'
  ORDER BY p.geometry <-> ref.geometry LIMIT 1

Range – count POIs within radius:
  SELECT COUNT(*) AS count
  FROM pois p
  WHERE ST_DWithin(p.geometry, (SELECT geometry FROM pois WHERE poi_name ILIKE '%<ref>%' LIMIT 1), <metres>)
    AND p.<category_col> ILIKE '%<value>%'
